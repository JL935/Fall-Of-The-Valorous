//REGION Upgrading
//when someone obtains an upgradable item, keep track of that
IF
TemplateAddedTo((ITEMROOT)_ValorItemTemplate, (ITEM)_ValorItem, (CHARACTER)_ValorCharacter, _)
AND
DB_GLO_FOTV_ItemUpgradeFlagSetPair(_, _ValorItemTemplate, _)
THEN
DB_FOTV_CharacterHasUpgradableItem(_ValorItemTemplate, _ValorItem, _ValorCharacter);

//when someone loses an upgradable item, also keep track of that
IF
TemplateRemovedFrom((ITEMROOT)_ValorItemTemplate, (ITEM)_ValorItem, (CHARACTER)_ValorCharacter)
AND
DB_GLO_FOTV_ItemUpgradeFlagSetPair(_, _ValorItemTemplate, _)
AND
DB_FOTV_CharacterHasUpgradableItem(_ValorItemTemplate, _ValorItem, _ValorCharacter)
THEN
NOT DB_FOTV_CharacterHasUpgradableItem(_ValorItemTemplate, _ValorItem, _ValorCharacter);

//When the condition is met to upgrade the gear, run PROC
IF
FlagSet(_Flag, _, _)
AND
DB_GLO_FOTV_ItemUpgradeFlagSetPair(_Flag, _, _ValorItemTemplateUpgrade)
THEN
PROC_FOTV_UpgradeItem(_ValorItemTemplateUpgrade);

//If the item to be upgraded is currently equipped, destroy it and directly equip the new one
PROC
PROC_FOTV_UpgradeItem((ITEMROOT)_ValorItemTemplateUpgrade)
AND
DB_FOTV_CharacterHasUpgradableItem(_ValorItemTemplate, _ValorItem, _ValorCharacter)
AND
DB_GLO_FOTV_ItemUpgradeFlagSetPair(_, _ValorItemTemplate, _ValorItemTemplateUpgrade)
AND
IsEquipped(_ValorItem, 1)
THEN
DB_FOTV_OldGearWasEquipped(_ValorItem);
RequestDelete(_ValorItem);
TemplateAddTo(_ValorItemTemplateUpgrade, _ValorCharacter, 1, 1);

//secondary step to go from added to equipped cuz yk
IF
TemplateAddedTo((ITEMROOT)_ValorItemTemplateUpgrade, _ValorItemUpgrade, _ValorCharacter, _)
AND
DB_GLO_FOTV_ItemUpgradeFlagSetPair(_Flag, _ValorDowngrade, _ValorItemTemplateUpgrade)
AND
DB_FOTV_OldGearWasEquipped((ITEM)_ValorItem)
THEN
Equip((CHARACTER)_ValorCharacter, (ITEM)_ValorItemUpgrade);
NOT DB_FOTV_OldGearWasEquipped((ITEM)_ValorItem);

//If the item to be upgraded wasn't equipped but in someone's inventory, destroy it and add the new one to their inventory
PROC
PROC_FOTV_UpgradeItem((ITEMROOT)_ValorItemTemplateUpgrade)
AND
DB_FOTV_CharacterHasUpgradableItem(_ValorItemTemplate, _ValorItem, _ValorCharacter)
AND
DB_GLO_FOTV_ItemUpgradeFlagSetPair(_Flag, _ValorItemTemplate, _ValorItemTemplateUpgrade)
AND
IsEquipped(_ValorItem, 0)
AND
ItemIsInUserInventory(_ValorItem, _ValorCharacter, 0, 1)
THEN
RequestDelete(_ValorItem);
TemplateAddTo(_ValorItemTemplateUpgrade, _ValorCharacter, 1, 1);

//If the item is in no one's inventory and has met the requirements of upgrading, destroy old upon adding to inventory and replace with new
IF
TemplateAddedTo((ITEMROOT)_ValorItemTemplate, (ITEM)_ValorItem, _ValorCharacter, _)
AND
DB_GLO_FOTV_ItemUpgradeFlagSetPair(_Flag, _ValorItemTemplate, _ValorItemTemplateUpgrade)
AND
GetFlag(_Flag, NULL_00000000-0000-0000-0000-000000000000, 1)
THEN
RequestDelete(_ValorItem);
TemplateAddTo(_ValorItemTemplateUpgrade, _ValorCharacter, 1, 1);
//END_REGION

AND
HasActiveStatus(_Object, _OldDamage, 1)
THEN
ApplyStatus(_Object, _NewDamage, -1.0);
