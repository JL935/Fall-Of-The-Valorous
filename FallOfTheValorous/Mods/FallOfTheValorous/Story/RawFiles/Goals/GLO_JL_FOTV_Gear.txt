//REGION Upgrading Tier 1 --> Tier 2
//When the condition is met to upgrade the gear, run PROC
IF
FlagSet(_Flag, _, _)
AND
DB_GLO_FOTV_ItemUpgradeFlagSetPair(_Flag, _OldGear, _NewGear)
THEN
DB_FOTV_GearTier(_NewGear);
PROC_FOTV_UpgradeItem(_OldGear, _NewGear);

//If the item to be upgraded is currently equipped, destroy it and directly equip the new one
PROC
PROC_FOTV_UpgradeItem((ITEM)_OldGear, (ITEMROOT)_NewGear)
AND
IsEquipped(_OldGear, 1)
AND
GetOwner(_OldGear, _Character)
THEN
DB_FOTV_OldGearWasEquipped((ITEM)_OldGear);
RequestDelete(_OldGear);
TemplateAddTo(_NewGear, _Character, 1, 1);

//If the item to be upgraded wasn't equipped but in someone's inventory, destroy it and add the new one to their inventory
PROC
PROC_FOTV_UpgradeItem((ITEM)_OldGear, (ITEMROOT)_NewGear)
AND
IsEquipped(_OldGear, 0)
AND
GetOwner(_OldGear, _Character)
AND
ItemIsInUserInventory(_OldGear, _Character, 0, 1)
THEN
RequestDelete(_OldGear);
TemplateAddTo(_NewGear, _Character, 1, 1);

//If the item is in no one's inventory and has met the requirements of upgrading, destroy old upon adding to inventory and replace with new
IF
AddedTo((ITEM)_OldGear, _Player, _)
AND
DB_GLO_FOTV_ItemUpgradeFlagSetPair(_, (ITEM)_OldGear, (ITEMROOT)_NewGear)
AND
DB_FOTV_GearTier(_NewGear)
THEN
RequestDelete(_OldGear);
TemplateAddTo(_NewGear, _Player, 1, 1);

//If the upgrade prereqs are met and someone not in the party is holding it, nothing happens, so here's a fallback
IF
Equipped(_OldGear, _Character)
AND
DB_GLO_FOTV_ItemUpgradeFlagSetPair(_, (ITEM)_OldGear, (ITEMROOT)_NewGear)
AND
DB_FOTV_GearTier(_NewGear)
THEN
RequestDelete(_OldGear);
TemplateAddTo(_NewGear, _Character, 1, 1);

IF
TemplateAddedTo(_ObjectTemplate, _Item, _Character, _)
AND
DB_GLO_FOTV_ItemUpgradeFlagSetPair(_, (ITEM)_OldGear, (ITEMROOT)_ObjectTemplate)
AND
DB_FOTV_OldGearWasEquipped((ITEM)_OldGear)
THEN
Equip((CHARACTER)_Character, (ITEM)_Item);
NOT DB_FOTV_OldGearWasEquipped((ITEM)_OldGear);

IF
TemplateAddedTo(_ObjectTemplate, _Item, _Character, _)
AND
DB_GLO_FOTV_ItemUpgradeFlagSetPair(_, (ITEM)_OldGear, (ITEMROOT)_ObjectTemplate)
AND
NOT DB_FOTV_StoreTierTwoItem((ITEMROOT)_ObjectTemplate, (ITEM)_Item)
THEN
DB_FOTV_StoreTierTwoItem((ITEMROOT)_ObjectTemplate, (ITEM)_Item);
//END_REGION
