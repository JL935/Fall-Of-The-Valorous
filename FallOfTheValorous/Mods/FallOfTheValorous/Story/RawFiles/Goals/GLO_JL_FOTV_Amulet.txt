//REGION Amulet abilities
//When you cast Hallowed Ground, all applicable enemies get a status that reduces their attacks against you while in the aura
IF
CastSpell(_Player, _HallowedGroundSpell, _, _, _)
AND
DB_GLO_FOTV_HallowedGroundSpell(_HallowedGroundSpell)
AND
DB_GLO_FOTV_HallowedGroundEnemyTag(_EnemyTag)
AND
DB_Is_InCombat(_Enemy, _)
AND
IsTagged(_Enemy, _EnemyTag, 1)
THEN
ApplyStatus(_Enemy, "JL_FOTV_HALLOWEDGROUND_MONSTROUS_PENALTIES", -1.0);
DB_GLO_FOTV_HallowedGroundCast(_Player);

//If an applicable enemy joins combat while the Hallowed Ground already exists, give them the penalty status
IF
EnteredCombat(_Enemy, _)
AND
DB_GLO_FOTV_HallowedGroundCast(_Player)
AND
DB_GLO_FOTV_HallowedGroundEnemyTag(_EnemyTag)
AND
IsTagged(_Enemy, _EnemyTag, 1)
THEN
ApplyStatus(_Enemy, "JL_FOTV_HALLOWEDGROUND_MONSTROUS_PENALTIES", -1.0);

//When someone in the Hallowed Ground takes damage of a certain type, add it to a DB then run a PROC to give resistance against it
IF
AttackedBy(_Defender, _, _, _DamageType, _DamageAmount, _, _)
AND
HasActiveStatus(_Defender, "JL_FOTV_HALLOWEDGROUND_BONUSES_TIER3", 1)
AND
_DamageAmount > 0
AND
DB_GLO_FOTV_DamageTypeResistancePair(_DamageType, _)
THEN
DB_GLO_FOTV_HallowedResistance(_DamageType);
PROC_FOTV_HallowedResistance();

//Run the resistance applier PROC when someone enters the Hallowed Ground
IF
StatusApplied(_Player, "JL_FOTV_HALLOWEDGROUND_BONUSES_TIER3", _, _)
THEN
DB_FOTV_IsInTierThreeHallow(_Player);
PROC_FOTV_HallowedResistance();

//if u leave T3 hallowed ground, get removed from the DB that says ur currently in it (why did i do this as a DB)
IF
StatusRemoved(_Player, "JL_FOTV_HALLOWEDGROUND_BONUSES_TIER3", _, _)
AND
DB_FOTV_IsInTierThreeHallow(_Player)
THEN
RealtimeObjectTimerLaunch(_Player, "JL_FOTV_HallowedGroundDespawn", 500);

//i don't remember why it needed to be a timer do i trust past me who didn't comment about it
IF
ObjectTimerFinished((CHARACTER)_Player, "JL_FOTV_HallowedGroundDespawn")
THEN
NOT DB_FOTV_IsInTierThreeHallow(_Player);

//Clear the DBs for tracking stuff when the Hallowed Ground exists (the status is on the hallowed ground itself)
IF
StatusRemoved(_, "JL_FOTV_HALLOWEDGROUND_MONSTROUS_BUFFS_TIMER", _, _)
AND
DB_GLO_FOTV_HallowedGroundCast(_Player)
AND
DB_GLO_FOTV_HallowedResistance(_DamageType)
THEN
NOT DB_GLO_FOTV_HallowedGroundCast(_Player);
NOT DB_GLO_FOTV_HallowedResistance(_DamageType);

//what
IF
StatusRemoved(_, "JL_FOTV_HALLOWEDGROUND_MONSTROUS_BUFFS_TIMER", _, _)
AND
DB_FOTV_IsInTierThreeHallow(_Player)
THEN
ApplyStatus(_Player, "JL_FOTV_HALLOWEDGROUND_CLEANSE", 6.0);

//Give resistance to everyone in Tier 3 Hallowed Ground to all damage types in the DB
PROC
PROC_FOTV_HallowedResistance()
AND
DB_Players(_Player)
AND
DB_GLO_FOTV_HallowedResistance(_DamageType)
AND
DB_GLO_FOTV_DamageTypeResistancePair(_DamageType, _ResistanceStatus)
AND
HasActiveStatus(_Player, "JL_FOTV_HALLOWEDGROUND_BONUSES_TIER3", 1)
THEN
ApplyStatus(_Player, _ResistanceStatus, -1.0);
//END_REGION

IF

THEN

//and u can try again tomorrow

NOT DB_FOTV_PeriaptCurseCheck(_Character);
