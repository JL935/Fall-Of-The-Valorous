//When entering Ravener's Desecration, force turn-based mode
IF
StatusApplied((GUIDSTRING)_Object, _DesecrationStatus, _, _)
AND
DB_RavenerDesecrationDamageStatus(_DesecrationStatus)
THEN
ForceTurnBasedMode((CHARACTER)_Object, 1);

//Dispelling Ravener's Desecration
IF
AttackedBy(_Desecrated, _, _, "Radiant", _DamageAmount, _, _)
AND
DB_FOTV_DealRadiantToDispel(_Desecrated, _RequiredAmount, _EvilStatus)
AND
_DamageAmount >= _RequiredAmount
THEN
RemoveStatus(_Desecrated, _EvilStatus);

//Disabling soul ward
IF
StatusApplied((GUIDSTRING)_Object, "JL_FOTV_RAVENER_SOULWARD_DISABLED", _, _)
AND
DB_FOTV_SoulWardStatus(_SoulWardStatus)
AND
HasActiveStatus(_Object, _SoulWardStatus, 1)
THEN
DB_FOTV_RavenerWardReturn(_Object, _SoulWardStatus);
RemoveStatus((GUIDSTRING)_Object, (STRING)_SoulWardStatus);

//Enabling soul ward
IF
StatusRemoved((GUIDSTRING)_Object, "JL_FOTV_RAVENER_SOULWARD_DISABLED", _, _)
AND
DB_FOTV_RavenerWardReturn(_Object, _SoulWardStatus)
THEN
ApplyStatus((GUIDSTRING)_Object, _SoulWardStatus, -1.0);
NOT DB_FOTV_RavenerWardReturn(_Object, _SoulWardStatus);

//Reveal the relevant passive on all raveners when you read the appropriate lore document
IF
FlagSet(_RavenerKnowledgeFlag, _, _)
AND
DB_RavenerKnowledgeFlag(_RavenerKnowledgeFlag, _AgeTag, _KnowledgePassive)
AND
DB_FOTV_Ravener(_Ravener)
AND
IsTagged(_Ravener, _AgeTag, 1)
THEN
AddPassive(_Ravener, _KnowledgePassive);

//Clear the block on dead inquisitors when their ravener is dead
IF
Died(_Ravener)
AND
DB_FOTV_SoulConsumers(_Ravener, _Inquisitor)
THEN
RemoveStatus(_Inquisitor, "JL_FOTV_RAVENER_CONSUMPTION_BLOCK_VAELYTHRA");
