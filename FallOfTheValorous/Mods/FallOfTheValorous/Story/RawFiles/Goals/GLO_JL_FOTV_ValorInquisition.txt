//REGION Unlocking Valor
//3 prereqs - defeat Taarogus, read some books, complete at least 1 Feat of Renown

//1. Taarogus defeated is the last step you complete
IF
QuestUpdateUnlocked(_, "QST_JL_FOTV_Act1", "STEP_JL_FOTV_ApostleCompleted")
AND
QRY_FOTV_HasReadEnoughLore()
AND
DB_GlobalCounter("JL_FOTV_FoR_TotalCompletions",_Count)
AND
_Count >= 1
THEN
SetFlag((FLAG)JL_FOTV_ValorSubclassUnlocked_30d714d5-35e8-458c-9a74-31dc9ccc512e);

//2. Reading lore is the last step you complete
IF
FlagSet(_, _, _)
AND
QRY_FOTV_HasReadEnoughLore()
AND
QRY_QuestUpdateIsUnlockedForAny("QST_JL_FOTV_Act1","STEP_JL_FOTV_ApostleCompleted")
AND
DB_GlobalCounter("JL_FOTV_FoR_TotalCompletions",_Count)
AND
_Count >= 1
AND
QRY_OnlyOnce("JL_FOTV_UnlockValorByLore")
THEN
SetFlag((FLAG)JL_FOTV_ValorSubclassUnlocked_30d714d5-35e8-458c-9a74-31dc9ccc512e);

//3. Completing a Feat of Renown is the last step you complete
IF
DB_GlobalCounter("JL_FOTV_FoR_TotalCompletions",_Count)
AND
_Count >= 1
AND
QRY_FOTV_HasReadEnoughLore()
AND
QRY_QuestUpdateIsUnlockedForAny("QST_JL_FOTV_Act1","STEP_JL_FOTV_ApostleCompleted")
AND
QRY_OnlyOnce("JL_FOTV_UnlockValorByRenown")
THEN
SetFlag((FLAG)JL_FOTV_ValorSubclassUnlocked_30d714d5-35e8-458c-9a74-31dc9ccc512e);
