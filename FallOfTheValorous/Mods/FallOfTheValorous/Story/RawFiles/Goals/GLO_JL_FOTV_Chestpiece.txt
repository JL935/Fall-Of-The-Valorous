//REGION Holy site tracking/flag setting
IF
EnteredTrigger((CHARACTER)_Character, (TRIGGER)_HSTrigger)
AND
DB_GLO_FOTV_HolySiteTriggers(_HSTrigger, _HSFlag)
AND
DB_Players(_Character)
AND
DB_Avatars(_Avatar)
THEN
SetFlag(_HSFlag);
PROC_IncreaseCounter("JL_FOTV_HolySitesVisited");
ApplyStatus(_Avatar, "JL_FOTV_HOLYSITE_ENTER_APPLYVFX", 0.0);
NOT DB_GLO_FOTV_HolySiteTriggers(_HSTrigger, _HSFlag);

IF
StatusApplied((CHARACTER)_Object, (STRING)_HSStatus, _, _)
AND
DB_GLO_FOTV_HolySiteStatuses(_HSStatus, _HSFlag)
AND
DB_Players(_Object)
AND
DB_Avatars(_Avatar)
THEN
SetFlag(_HSFlag);
PROC_IncreaseCounter("JL_FOTV_HolySitesVisited");
ApplyStatus(_Avatar, "JL_FOTV_HOLYSITE_ENTER_APPLYVFX", 0.0);
NOT DB_GLO_FOTV_HolySiteStatuses(_HSStatus, _HSFlag);

IF
GameBookInterfaceClosed(S_JL_FOTV_CahootShrine_66395e14-9a44-4360-88fe-6efe5f75e7f8, _)
AND
QRY_OnlyOnce("JL_FOTV_CahootPaidRespects")
AND
DB_Avatars(_Avatar)
THEN
SetFlag((FLAG)JL_FOTV_VisitedHolySite_Cahoot_2c671fdf-ce4d-42ec-9ef6-6ecc196f1eb1);
PROC_IncreaseCounter("JL_FOTV_HolySitesVisited");
ApplyStatus(_Avatar, "JL_FOTV_HOLYSITE_ENTER_APPLYVFX", 0.0);
PROC_HidePartyMarker("MARKER_JL_FOTV_BootyCahooty");
//END_REGION

PROC_UnlockPartyMarker("MARKER_JL_FOTV_ValorousVirtues");
PROC_UnlockPartyMarker("MARKER_JL_FOTV_BootyCahooty");

//set flag for finishing VV subquest
IF
QuestUpdateUnlocked(_, "QST_JL_FOTV_BecomeHoly", "STEP_JL_FOTV_IsRenowned")
THEN
SetFlag(JL_FOTV_Chestpiece_IsRenowned_13b3661e-6f55-4e57-a2ae-21f641afa6cf);

THEN
SetFlag(JL_FOTV_Chestpiece_IsItinerant_72d8ad21-b730-4091-8374-78659d053347);

//advance main quest when both subquests completed
IF
DB_GlobalFlag((FLAG)JL_FOTV_Chestpiece_IsRenowned_13b3661e-6f55-4e57-a2ae-21f641afa6cf)
AND
DB_GlobalFlag((FLAG)JL_FOTV_Chestpiece_IsItinerant_72d8ad21-b730-4091-8374-78659d053347)
THEN
QuestUpdate("QST_JL_FOTV_BecomeHoly", "STEP_JL_FOTV_BreakTheAegis");
PROC_UnlockPartyMarker("MARKER_JL_FOTV_DestroyAegis");

//grant the spell to break the aegis when you're close to it and have completed a feat of renown
IF
StatusApplied((CHARACTER)_Player, "JL_FOTV_NEARMANTLE", _, _)
AND
DB_Players(_Player)
AND
GetFlag(JL_FOTV_Chestpiece_IsRenowned_13b3661e-6f55-4e57-a2ae-21f641afa6cf, NULL_00000000-0000-0000-0000-000000000000, 1)
THEN
AddSpell(_Player, "JL_FOTV_Projectile_Fly_EmpyrealMantle");

//remove the breaky spell when far from chest
IF
StatusRemoved((CHARACTER)_Player, "JL_FOTV_NEARMANTLE", _, _)
AND
DB_Players(_Player)
AND
HasSpell(_Player, "JL_FOTV_Projectile_Fly_EmpyrealMantle", 1)
THEN
RemoveSpell(_Player, "JL_FOTV_Projectile_Fly_EmpyrealMantle");

//break the aegis when u cast the spell
IF
CastSpell(_, "JL_FOTV_Projectile_Fly_EmpyrealMantle", _, _, _)
THEN
RemoveStatus(S_GEAR_JL_FOTV_EmpyrealMantle_f269dc0a-183b-4454-b9bc-22dc8370d7f7, "JL_FOTV_EMPYREALMANTLE_HOLYWARD");
SetCanInteract(S_GEAR_JL_FOTV_EmpyrealMantle_f269dc0a-183b-4454-b9bc-22dc8370d7f7, 1);
SetOnStage(S_JL_FOTV_EmpyrealMantle_HolyWard_c19d1652-3282-416c-8ab5-b6f96b7b1f47, 0);
QuestClose("QST_JL_FOTV_BecomeHoly");
//END_REGION

//set the quest steps for any VV already completed
IF
QuestUpdateUnlocked(_, "QST_JL_FOTV_BecomeHoly", "STEP_JL_FOTV_GoDoRenownThings")
AND
DB_GLO_FOTV_VV_Flag_QS(_VVCounter, _VVFlag, _VVStep)
AND
GetFlag(_VVFlag, NULL_00000000-0000-0000-0000-000000000000, 1)
THEN
QuestUpdate("QST_JL_FOTV_BecomeHoly", _VVStep);
NOT DB_GLO_FOTV_VV_Flag_QS(_VVCounter, _VVFlag, _VVStep);

//set the flag for completing a VV
IF
DB_GlobalCounter(_VVCounter, _Count)
AND
DB_GLO_FOTV_VV_Flag_QS(_VVCounter, _VVFlag, _)
AND
_Count >= 1
THEN
SetFlag(_VVFlag);

//if you haven't already unlocked the relevant step, do so when the flag is set for completing ur first VV of that type
IF
FlagSet(_VVFlag, _, _)
AND
DB_GLO_FOTV_VV_Flag_QS(_VVCounter, _VVFlag, _VVStep)
AND
QRY_QuestUpdateIsUnlockedForAny("QST_JL_FOTV_BecomeHoly", "STEP_JL_FOTV_GoDoRenownThings")
AND
NOT QRY_QuestUpdateIsUnlockedForAny("QST_JL_FOTV_BecomeHoly", _VVStep)
THEN
QuestUpdate("QST_JL_FOTV_BecomeHoly", _VVStep);
NOT DB_GLO_FOTV_VV_Flag_QS(_VVCounter, _VVFlag, _VVStep);

//finish subquest once u finish any FoR
IF
DB_GlobalCounter("JL_FOTV_FoR_TotalCompletions",_Count)
AND
_Count >= 1
AND
QRY_QuestUpdateIsUnlockedForAny("QST_JL_FOTV_BecomeHoly", "STEP_JL_FOTV_GoDoRenownThings")
AND
NOT QRY_QuestUpdateIsUnlockedForAny("QST_JL_FOTV_BecomeHoly", "STEP_JL_FOTV_IsRenowned")
THEN
QuestUpdate("QST_JL_FOTV_BecomeHoly", "STEP_JL_FOTV_FinishedRenown");
//END_REGION
